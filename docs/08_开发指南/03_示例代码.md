    // 执行业务逻辑
    // ...
    
    // 关闭AuroraRT

    std::cout << "AuroraRT shut down successfully" << std::endl;
    
    return 0;
}

## 2. 通信模式示例

### 2.1 发布-订阅模式

#### 2.1.1 发布者示例

```cpp
#include <aurorart/communication/pub_sub_pattern.h>
#include <iostream>
#include <string>

int main() {
    // 初始化AuroraRT
    aurorart::init();
    
    // 创建发布-订阅模式实例
    auto pubSubPattern = std::make_shared<aurorart::communication::PubSubPattern>();
    pubSubPattern->init();
    pubSubPattern->start();
    
    // 创建发布者
    auto publisher = pubSubPattern->createPublisher<std::string>("test_topic");
    
    // 发布消息
    for (int i = 0; i < 10; ++i) {
        std::string message = "Hello, AuroraRT! Message " + std::to_string(i);
        publisher->publish(&message, sizeof(message));
        std::cout << "Published: " << message << std::endl;
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
    }
    
    // 停止发布-订阅模式
    pubSubPattern->stop();
    
    // 关闭AuroraRT
    aurorart::shutdown();
    
    return 0;
}
```

## 4. 域管理示例

### 4.1 域和分区管理

```cpp
#include <aurorart/domain/domain_manager.h>
#include <iostream>

int main() {
    // 初始化AuroraRT
    aurorart::init();
    
    // 初始化域管理器
    aurorart::domain::DomainManager::instance().init();
    
    // 创建域
    bool domainCreated = aurorart::domain::DomainManager::instance().createDomain("test_domain", "Test domain");
    if (domainCreated) {
        std::cout << "Domain created successfully" << std::endl;
    }
    
    // 创建分区
    bool partitionCreated = aurorart::domain::DomainManager::instance().createPartition("test_domain", "test_partition", "Test partition");
    if (partitionCreated) {
        std::cout << "Partition created successfully" << std::endl;
    }
    
    // 加入域
    bool joinedDomain = aurorart::domain::DomainManager::instance().joinDomain("test_domain", "test_node");
    if (joinedDomain) {
        std::cout << "Node joined domain successfully" << std::endl;
    }
    
    // 加入分区
    bool joinedPartition = aurorart::domain::DomainManager::instance().joinPartition("test_domain", "test_partition", "test_node");
    if (joinedPartition) {
        std::cout << "Node joined partition successfully" << std::endl;
    }
    
    // 列出域
    auto domains = aurorart::domain::DomainManager::instance().listDomains();
    std::cout << "Domains:" << std::endl;
    for (const auto& domain : domains) {
        std::cout << "  - " << domain << std::endl;
    }
    
    // 列出分区
    auto partitions = aurorart::domain::DomainManager::instance().listPartitions("test_domain");
    std::cout << "Partitions in test_domain:" << std::endl;
    for (const auto& partition : partitions) {
        std::cout << "  - " << partition << std::endl;
    }
    
    // 离开分区
    bool leftPartition = aurorart::domain::DomainManager::instance().leavePartition("test_domain", "test_partition", "test_node");
    if (leftPartition) {
        std::cout << "Node left partition successfully" << std::endl;
    }
    
    // 离开域
    bool leftDomain = aurorart::domain::DomainManager::instance().leaveDomain("test_domain", "test_node");
    if (leftDomain) {
        std::cout << "Node left domain successfully" << std::endl;
    }
    
    // 删除分区
    bool partitionDeleted = aurorart::domain::DomainManager::instance().deletePartition("test_domain", "test_partition");
    if (partitionDeleted) {
        std::cout << "Partition deleted successfully" << std::endl;
    }
    
    // 删除域
    bool domainDeleted = aurorart::domain::DomainManager::instance().deleteDomain("test_domain");
    if (domainDeleted) {
        std::cout << "Domain deleted successfully" << std::endl;
    }
    
    // 关闭域管理器
    aurorart::domain::DomainManager::instance().shutdown();
    
    // 关闭AuroraRT
    aurorart::shutdown();
    
    return 0;
}
```

## 6. 热重启示例

### 6.1 热重启使用

```cpp
#include <aurorart/hot_restart/hot_restart_manager.h>
#include <iostream>

int main() {
    // 初始化AuroraRT
    aurorart::init();
    
    // 配置热重启
    aurorart::hot_restart::HotRestartConfig config;
    config.stateSaving.enabled = true;
    config.stateSaving.saveConfig = true;
    config.stateSaving.saveRuntimeState = true;
    config.restartPolicy.useHotRestart = true;
    config.restartPolicy.maxRestartAttempts = 3;
    
    // 应用配置
    aurorart::hot_restart::HotRestartManager::instance().setConfig(config);
    
    // 注册状态保存回调
    aurorart::hot_restart::StateSaver::instance().registerSaveCallback([](aurorart::hot_restart::StateBuffer& buffer) {
        // 保存自定义状态
        int customValue = 42;
        buffer.save("custom_value", customValue);
        std::cout << "Custom state saved: " << customValue << std::endl;
    });
    
    // 注册状态恢复回调
    aurorart::hot_restart::StateRestorer::instance().registerRestoreCallback([](const aurorart::hot_restart::StateBuffer& buffer) {
        // 恢复自定义状态
        int customValue;
        if (buffer.load("custom_value", customValue)) {
            std::cout << "Custom state restored: " << customValue << std::endl;
        }
    });
    
    // 注册热重启事件回调
    aurorart::hot_restart::HotRestartManager::instance().registerEventCallback([](const aurorart::hot_restart::HotRestartEvent& event) {
        std::cout << "Hot restart event: " << event.getType() << std::endl;
    });
    
    // 触发热重启
    std::cout << "Triggering hot restart..." << std::endl;
    bool success = aurorart::hot_restart::HotRestartManager::instance().restartNode("test_node");
    if (success) {
        std::cout << "Hot restart triggered successfully" << std::endl;
    } else {
        std::cout << "Failed to trigger hot restart" << std::endl;
    }
    
    // 关闭AuroraRT
    aurorart::shutdown();
    
    return 0;
}
```

        publisher->publish(&message, sizeof(message));
        std::cout << "Published: " << message << std::endl;
        std::this_thread::sleep_for(std::chrono::milliseconds(500));
    }
    
    // 收集监控指标
    auto metrics = monitor->collectMetrics();
    std::cout << "\nCollected metrics:" << std::endl;
    for (const auto& metric : metrics) {
        std::cout << "  - " << metric.getName() << ": " << metric.getValue() << std::endl;
    }
    
    // 停止组件
    subscriber.reset();
    publisher.reset();
    pubSubPattern->stop();
    
    monitor->stop();
    aurorart::monitor::MonitorManager::instance().stop();
    
    node->stop();
    aurorart::node::NodeManager::instance().stop();
    
    // 关闭AuroraRT
    aurorart::shutdown();
    std::cout << "\nAuroraRT shut down successfully" << std::endl;
    
    return 0;
}
